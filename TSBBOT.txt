local function clearEnemyBackstep()
    enemyBackstepUntil = 0
    enemyBackstepTarget = nil

local function activateEnemyBackstep(enemyPlayer)
    if enemyPlayer and enemyPlayer == currentEnemy then
        enemyBackstepTarget = enemyPlayer
        enemyBackstepUntil = tick() + CFG.enemyBackstepWindow
        enemyBackstepCloseMode = false
        enemyBackstepAdvanceAttack = false
        enemyBackstepActionUntil = 0
        local localRoot = character and character:FindFirstChild("HumanoidRootPart")
        local targetRoot = currentEnemyChar and currentEnemyChar:FindFirstChild("HumanoidRootPart")
        if localRoot and targetRoot
            and (targetRoot.Position - localRoot.Position).Magnitude <= CFG.enemyBackstepTriggerRange then
            enemyBackstepCloseMode = true
            -- 动画命中瞬间停止普通连击，避免旧攻击协程再补一拳。
            isAttacking = false
            fireServer({ { Goal = "LeftClickRelease", Mobile = true } })
        end
    end
end

local function isEnemyBackstepActive(targetRoot)
    return targetRoot ~= nil
        and targetRoot.Parent ~= nil
        and enemyBackstepTarget ~= nil
        and enemyBackstepTarget == currentEnemy
        and tick() <= enemyBackstepUntil
end

local function getEnemyBackstepPoint(targetRoot, distance)
    if not targetRoot or not targetRoot.Parent then return nil end
    local flatLook = Vector3.new(targetRoot.CFrame.LookVector.X, 0, targetRoot.CFrame.LookVector.Z)
    if flatLook.Magnitude <= 0.05 then return nil end
    return targetRoot.Position - flatLook.Unit * (distance or CFG.enemyBackstepDistance)
end

local function getRandomBackFanCFrame(targetRoot)
    if not targetRoot or not targetRoot.Parent then return nil end

    local flatLook = Vector3.new(targetRoot.CFrame.LookVector.X, 0,
        targetRoot.CFrame.LookVector.Z)
    local flatRight = Vector3.new(targetRoot.CFrame.RightVector.X, 0,
        targetRoot.CFrame.RightVector.Z)
    if flatLook.Magnitude <= 0.05 or flatRight.Magnitude <= 0.05 then return nil end

    local minRadius = math.max(0, math.min(CFG.selfBackFanMinRadius,
        CFG.selfBackFanMaxRadius))
    local maxRadius = math.max(minRadius, CFG.selfBackFanMaxRadius)
    local radius = minRadius
    if maxRadius > minRadius then
        radius = math.sqrt(rng:NextNumber(minRadius * minRadius,
            maxRadius * maxRadius))
    end
    local angle = math.rad(rng:NextNumber(-CFG.selfBackFanHalfAngle,
        CFG.selfBackFanHalfAngle))
    local direction = -flatLook.Unit * math.cos(angle)
        + flatRight.Unit * math.sin(angle)
    local position = targetRoot.Position + direction * radius
    local lookPoint = Vector3.new(targetRoot.Position.X, position.Y,
        targetRoot.Position.Z)
    return CFrame.lookAt(position, lookPoint)
end

local function getEnemyFacingDot(localRoot, targetRoot)
    if not localRoot or not targetRoot then return nil end
    local flatLook = Vector3.new(targetRoot.CFrame.LookVector.X, 0, targetRoot.CFrame.LookVector.Z)
    local targetToLocal = Vector3.new(
        localRoot.Position.X - targetRoot.Position.X,
        0,
        localRoot.Position.Z - targetRoot.Position.Z
    )
    if flatLook.Magnitude <= 0.05 or targetToLocal.Magnitude <= 0.05 then return nil end
    return flatLook.Unit:Dot(targetToLocal.Unit)
end

local function isOutsideEnemyFront(localRoot, targetRoot, keepRearAngle)
    local facingDot = getEnemyFacingDot(localRoot, targetRoot)
    if facingDot == nil then return false end
    local threshold = keepRearAngle and CFG.enemyBackstepRearExitDot
        or CFG.enemyBackstepRearEnterDot
    return facingDot <= threshold
end

local function getRadialPoint(targetRoot, localRoot, distance)
    if not targetRoot or not localRoot then return nil end
    local radial = Vector3.new(
        localRoot.Position.X - targetRoot.Position.X,
        0,
        localRoot.Position.Z - targetRoot.Position.Z
    )
    if radial.Magnitude <= 0.05 then
        radial = -Vector3.new(targetRoot.CFrame.LookVector.X, 0,
            targetRoot.CFrame.LookVector.Z)
    end
    if radial.Magnitude <= 0.05 then return nil end
    return targetRoot.Position + radial.Unit * distance
end

local function getSideDashPlacementPoint(targetRoot, localRoot, tryBackstep, attach)
    if tryBackstep then
        local distance = attach and CFG.sideDashBackAttachDistance
            or CFG.enemyBackstepSideDashTargetDistance
        return getEnemyBackstepPoint(targetRoot, distance)
    end
    return getRadialPoint(targetRoot, localRoot, CFG.enemyBackstepAttackDistance)
end

local function applySideDashAttachment(localRoot, targetRoot, tryBackstep, maxStep)
    if not localRoot or not localRoot.Parent or not targetRoot or not targetRoot.Parent then
        return false
    end

    local desiredPoint = getSideDashPlacementPoint(targetRoot, localRoot, tryBackstep, true)
    if not desiredPoint then return false end

    local delta = Vector3.new(
        desiredPoint.X - localRoot.Position.X,
        0,
        desiredPoint.Z - localRoot.Position.Z
    )
    if delta.Magnitude <= CFG.sideDashAttachTolerance then return false end

    local nextPosition = localRoot.Position + delta.Unit
        * math.min(delta.Magnitude, maxStep or CFG.sideDashBackAttachMaxStep)
    local lookPoint = Vector3.new(targetRoot.Position.X, nextPosition.Y, targetRoot.Position.Z)
    localRoot.CFrame = CFrame.lookAt(nextPosition, lookPoint)
    camera.CFrame = CFrame.lookAt(camera.CFrame.Position, targetRoot.Position)
    return true
end

local function applySideDashBackSlide(localRoot, targetRoot)
    if not localRoot or not localRoot.Parent or not targetRoot or not targetRoot.Parent then
        return false
    end

    local startedAt = tick()
    local travelLeft = CFG.sideDashBackSlideMaxTravel
    local moved = false
    while runtime.alive and mode == "combat" and not isStandby
        and localRoot.Parent and targetRoot.Parent
        and travelLeft > 0.05
        and tick() - startedAt < CFG.sideDashBackSlideDuration do
        local desiredPoint = getEnemyBackstepPoint(targetRoot,
            CFG.sideDashBackAttachDistance)
        if not desiredPoint then break end

        local delta = Vector3.new(
            desiredPoint.X - localRoot.Position.X,
            0,
            desiredPoint.Z - localRoot.Position.Z
        )
        if delta.Magnitude <= CFG.sideDashAttachTolerance then break end

        local step = math.min(delta.Magnitude, travelLeft, CFG.sideDashBackSlideStep)
        local nextPosition = localRoot.Position + delta.Unit * step
        local lookPoint = Vector3.new(targetRoot.Position.X, nextPosition.Y,
            targetRoot.Position.Z)
        localRoot.CFrame = CFrame.lookAt(nextPosition, lookPoint)
        humanoid:Move(delta.Unit, false)
        camera.CFrame = CFrame.lookAt(camera.CFrame.Position, targetRoot.Position)
        travelLeft = travelLeft - step
        moved = true
        RunService.Heartbeat:Wait()
    end
    return moved
end

local function moveAroundEnemyAtDistance(localRoot, targetRoot, distance, separation)
    if not localRoot or not targetRoot then return end
    local radial = Vector3.new(
        localRoot.Position.X - targetRoot.Position.X,
        0,
        localRoot.Position.Z - targetRoot.Position.Z
    )
    local flatLook = Vector3.new(targetRoot.CFrame.LookVector.X, 0,
        targetRoot.CFrame.LookVector.Z)
    if radial.Magnitude <= 0.05 or flatLook.Magnitude <= 0.05 then return end

    local radialUnit = radial.Unit
    local desiredBack = -flatLook.Unit
    local tangent = desiredBack - radialUnit * desiredBack:Dot(radialUnit)
    if tangent.Magnitude <= 0.05 then
        -- At exactly the enemy's front, desiredBack is collinear with radial.
        -- Pick a lateral direction so hold-distance also begins an orbit.
        tangent = Vector3.new(targetRoot.CFrame.RightVector.X, 0,
            targetRoot.CFrame.RightVector.Z)
    end
    local distanceError = distance - radial.Magnitude
    local moveDirection = radialUnit * math.clamp(distanceError / 3, -1, 1)
    if tangent.Magnitude > 0.05 then
        moveDirection = moveDirection + tangent.Unit
    end
    if separation.Magnitude > 0.05 then
        moveDirection = moveDirection + separation.Unit * 0.35
    end

    humanoid:Move(moveDirection.Magnitude > 0.05 and moveDirection.Unit
        or Vector3.zero, false)
end

local function isSideDashActionLocked()
    return tick() < sideDashActionUntil
end

local function isEnemyBackstepCloseActive(targetRoot)
    return enemyBackstepCloseMode and isEnemyBackstepActive(targetRoot)
end

local function isEnemyBackstepFrontBlocked()
    local localRoot = character and character:FindFirstChild("HumanoidRootPart")
    local targetRoot = currentEnemyChar and currentEnemyChar:FindFirstChild("HumanoidRootPart")
    return isEnemyBackstepCloseActive(targetRoot)
        and not isOutsideEnemyFront(localRoot, targetRoot, enemyBackstepAdvanceAttack)
end

local function performDash(targetRoot, tryBackstep)
    if tryBackstep and isEnemyBackstepActive(targetRoot) then
        local localRoot = character and character:FindFirstChild("HumanoidRootPart")
        local backPoint = getEnemyBackstepPoint(targetRoot)
        if localRoot and backPoint then
            local flatPoint = Vector3.new(backPoint.X, localRoot.Position.Y, backPoint.Z)
            if (flatPoint - localRoot.Position).Magnitude > 0.05 then
                pcall(function()
                    localRoot.CFrame = CFrame.lookAt(localRoot.Position, flatPoint)
                    camera.CFrame = CFrame.lookAt(camera.CFrame.Position, targetRoot.Position)
                end)
            end
        end
    end
    fireServer({ { Dash = Enum.KeyCode.W, Key = Enum.KeyCode.Q, Goal = "KeyPress" } })
end

local function performSideDash(targetRoot, tryBackstep, attackAfterDash)
    local now = tick()
    if now - lastSideDash < CFG.sideDashInterval then return false end
    lastSideDash = now
    local shouldAttackAfterDash = attackAfterDash == true
    local specialBackstep = isEnemyBackstepCloseActive(targetRoot)
    local attachDelay = tryBackstep
        and (CFG.sideDashBackAttachFollowUpDelay + CFG.sideDashBackSlideDuration)
        or 0
    local actionLockUntil = now + CFG.sideDashDuration + attachDelay
        + (shouldAttackAfterDash and CFG.attackPressTime + 0.12 or 0.08)
    sideDashActionUntil = math.max(sideDashActionUntil, actionLockUntil)
    if shouldAttackAfterDash then
        isAttacking = false
        fireServer({ { Goal = "LeftClickRelease", Mobile = true } })
        if specialBackstep then
            enemyBackstepActionUntil = math.max(enemyBackstepActionUntil,
                actionLockUntil)
        end
    end
    task.spawn(function()
        local localRoot = character and character:FindFirstChild("HumanoidRootPart")

        local function aimRightSideAtTarget()
            if not localRoot or not localRoot.Parent or not targetRoot or not targetRoot.Parent then
                return false
            end

            local aimPoint = getSideDashPlacementPoint(targetRoot, localRoot, tryBackstep, false)
                or targetRoot.Position
            local flatToTarget = Vector3.new(
                aimPoint.X - localRoot.Position.X,
                0,
                aimPoint.Z - localRoot.Position.Z
            )
            if flatToTarget.Magnitude <= 0.05 then return false end

            -- 角色的右方向对准敌人，使纯 D+Q 侧闪向敌人收近。
            local dashLook = Vector3.yAxis:Cross(flatToTarget.Unit)
            localRoot.CFrame = CFrame.lookAt(
                localRoot.Position,
                localRoot.Position + dashLook
            )
            camera.CFrame = CFrame.lookAt(camera.CFrame.Position, targetRoot.Position)
            return true
        end

        aimRightSideAtTarget()
        if VIM then
            pcall(function()
                humanoid:Move(Vector3.zero, false)
                VIM:SendKeyEvent(true,  Enum.KeyCode.D, false, game)
                VIM:SendKeyEvent(true,  Enum.KeyCode.Q, false, game)
            end)

            local startedAt = tick()
            while runtime.alive and tick() - startedAt < CFG.sideDashDuration do
                if mode ~= "combat" or isStandby or not trackEnabled
                    or not aimRightSideAtTarget() then
                    break
                end
                RunService.RenderStepped:Wait()
            end

            pcall(function()
                VIM:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
                VIM:SendKeyEvent(false, Enum.KeyCode.D, false, game)
            end)
        else
            fireServer({ { Dash = Enum.KeyCode.D, Key = Enum.KeyCode.Q, Goal = "KeyPress" } })
            task.wait(CFG.sideDashDuration)
        end

        if runtime.alive and mode == "combat" and not isStandby
            and localRoot and localRoot.Parent and targetRoot and targetRoot.Parent then
            applySideDashAttachment(localRoot, targetRoot, tryBackstep)
            if tryBackstep then
                task.wait(CFG.sideDashBackAttachFollowUpDelay)
                if runtime.alive and mode == "combat" and not isStandby
                    and localRoot.Parent and targetRoot.Parent then
                    applySideDashAttachment(localRoot, targetRoot, true,
                        CFG.sideDashBackAttachFollowUpMaxStep)
                    applySideDashBackSlide(localRoot, targetRoot)
                end
            end
            local flatTarget = Vector3.new(
                targetRoot.Position.X,
                localRoot.Position.Y,
                targetRoot.Position.Z
            )
            if (flatTarget - localRoot.Position).Magnitude > 0.05 then
                localRoot.CFrame = CFrame.lookAt(localRoot.Position, flatTarget)
            end
            local safeToAttack = not isEnemyBackstepFrontBlocked()
                and (not tryBackstep
                    or isOutsideEnemyFront(localRoot, targetRoot, true))
            if shouldAttackAfterDash and safeToAttack then
                fireServer({ { Goal = "LeftClick", Mobile = true } })
                task.wait(CFG.attackPressTime)
                fireServer({ { Goal = "LeftClickRelease", Mobile = true } })
            end
        end
    end)
    return true
end

local function getStandbyCFrame(mRoot)
    -- 目标：真正进入地下隐身挂机点，而不是卡在地面上方。
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = character and { character } or {}
    params.IgnoreWater = true

    local origin = mRoot.Position + Vector3.new(0, 25, 0)
    local result = workspace:Raycast(origin, Vector3.new(0, -500, 0), params)

    local groundY = mRoot.Position.Y
    if result then
        groundY = result.Position.Y
    elseif humanoid then
        -- 射线没命中时，用当前站立高度估算地面
        groundY = mRoot.Position.Y - (humanoid.HipHeight + (mRoot.Size.Y * 0.5))
    end

    local depth = math.max(tonumber(CFG.standbyDepth) or 120, 40)
    local hoverPosition = Vector3.new(
        mRoot.Position.X,
        groundY - depth,
        mRoot.Position.Z
    )
    return CFrame.new(hoverPosition) * mRoot.CFrame.Rotation
end

local function applyStandbyStealth(enable)
    if not character then return end

    if enable then
        standbySavedParts = {}
        for _, inst in ipairs(character:GetDescendants()) do
            if inst:IsA("BasePart") then
                standbySavedParts[inst] = {
                    CanCollide = inst.CanCollide,
                    CanQuery = inst.CanQuery,
                    CanTouch = inst.CanTouch,
                    Massless = inst.Massless,
                    Transparency = inst.Transparency,
                    LocalTransparencyModifier = inst.LocalTransparencyModifier,
                }
                if CFG.standbyNoclip ~= false then
                    inst.CanCollide = false
                    inst.CanQuery = false
                    inst.CanTouch = false
                    inst.Massless = true
                end
                if CFG.standbyLocalHide ~= false then
                    -- LocalTransparencyModifier 只影响本地渲染，退出时恢复
                    inst.LocalTransparencyModifier = 1
                end
            elseif inst:IsA("Decal") or inst:IsA("Texture") then
                standbySavedParts[inst] = {
                    Transparency = inst.Transparency,
                }
                if CFG.standbyLocalHide ~= false then
                    inst.Transparency = 1
                end
            elseif inst:IsA("ParticleEmitter") or inst:IsA("Trail") or inst:IsA("Beam") or inst:IsA("Fire") or inst:IsA("Smoke") or inst:IsA("Sparkles") then
                standbySavedParts[inst] = {
                    Enabled = inst.Enabled,
                }
                if CFG.standbyLocalHide ~= false then
                    inst.Enabled = false
                end
            end
        end

        if humanoid then
            standbyDisplayType = humanoid.DisplayDistanceType
            pcall(function()
                humanoid.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                humanoid.NameDisplayDistance = 0
                humanoid.HealthDisplayDistance = 0
            end)
        end

        -- 尽量隐藏默认名字/血条 Billboard
        pcall(function()
            local head = character:FindFirstChild("Head")
            if head then
                for _, gui in ipairs(head:GetChildren()) do
                    if gui:IsA("BillboardGui") then
                        standbySavedParts[gui] = { Enabled = gui.Enabled }
                        gui.Enabled = false
                    end
                end
            end
        end)
    else
        for inst, saved in pairs(standbySavedParts) do
            if inst and inst.Parent then
                pcall(function()
                    if inst:IsA("BasePart") then
                        if saved.CanCollide ~= nil then inst.CanCollide = saved.CanCollide end
                        if saved.CanQuery ~= nil then inst.CanQuery = saved.CanQuery end
                        if saved.CanTouch ~= nil then inst.CanTouch = saved.CanTouch end
                        if saved.Massless ~= nil then inst.Massless = saved.Massless end
                        if saved.Transparency ~= nil then inst.Transparency = saved.Transparency end
                        if saved.LocalTransparencyModifier ~= nil then
                            inst.LocalTransparencyModifier = saved.LocalTransparencyModifier
                        end
                    elseif inst:IsA("Decal") or inst:IsA("Texture") then
                        if saved.Transparency ~= nil then inst.Transparency = saved.Transparency end
                    elseif inst:IsA("ParticleEmitter") or inst:IsA("Trail") or inst:IsA("Beam") or inst:IsA("Fire") or inst:IsA("Smoke") or inst:IsA("Sparkles") then
                        if saved.Enabled ~= nil then inst.Enabled = saved.Enabled end
                    elseif inst:IsA("BillboardGui") then
                        if saved.Enabled ~= nil then inst.Enabled = saved.Enabled end
                    end
                end)
            end
        end
        standbySavedParts = {}

        if humanoid and standbyDisplayType ~= nil then
            pcall(function()
                humanoid.DisplayDistanceType = standbyDisplayType
            end)
        end
        standbyDisplayType = nil
    end
end

local function enterStandby()
    local mRoot = character and character:FindFirstChild("HumanoidRootPart")
    if not mRoot or not humanoid then return false end

    standbyAnchor = getStandbyCFrame(mRoot)
    standbyAutoRotate = humanoid.AutoRotate
    isStandby = true
    blitzJumpPending = false
    clearEnemyBackstep()
    cancelCounterAttack()
    stopAttacking()
    mode = "guard"
    currentEnemy = nil
    currentEnemyChar = nil

    pcall(function()
        applyStandbyStealth(true)
        lockStandbyPose(mRoot)
        -- 再写一次，抵消部分服务器/物理回写
        task.defer(function()
            if isStandby and mRoot and mRoot.Parent then
                pcall(function() lockStandbyPose(mRoot) end)
            end
        end)
    end)
    return true
end

local function exitStandby(destination)
    local mRoot = character and character:FindFirstChild("HumanoidRootPart")
    isStandby = false

    pcall(function()
        if mRoot then
            mRoot.AssemblyLinearVelocity = Vector3.zero
            mRoot.AssemblyAngularVelocity = Vector3.zero
            if destination then
                mRoot.CFrame = destination
            end
            mRoot.Anchored = false
        end
        applyStandbyStealth(false)
        if humanoid then
            humanoid.PlatformStand = false
            if standbyAutoRotate ~= nil then
                humanoid.AutoRotate = standbyAutoRotate
            else
                humanoid.AutoRotate = true
            end
            pcall(function()
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end)
        end
    end)

    standbyAnchor = nil
    standbyAutoRotate = nil
end

        or (ultimateActive and not ultimateSkillsReady) then

    end
    if arenaEnabled and not trainingEnabled and arenaRole ~= "fighter" then return false end
    local now = tick()

    local mRoot = character and character:FindFirstChild("HumanoidRootPart")
    local eRoot = currentEnemyChar and currentEnemyChar:FindFirstChild("HumanoidRootPart")
    if not mRoot or not eRoot then return false end

    performDash()
    return true

local function getAlivePlayerParts(player)

    local root = playerCharacter and playerCharacter:FindFirstChild("HumanoidRootPart")

local function getArenaMembers(masterPlayer)
    local members = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= masterPlayer and (player == LP or isWhitelisted(player)) then
            local root, playerHumanoid = getAlivePlayerParts(player)
            if root and playerHumanoid then
                table.insert(members, {
                    player = player,
                    root = root,
                    humanoid = playerHumanoid,
                    healthRatio = playerHumanoid.Health / math.max(playerHumanoid.MaxHealth, 1),
                })
            end
        end
    end
    table.sort(members, function(left, right)
        return left.player.Name < right.player.Name
    end)
    return members
end

local function distanceBucket(distance)
    return math.floor(distance * 10 + 0.5)
end

local function nearestThreatForMember(member, threats)
    local nearest, nearestBucket = nil, math.huge
    for _, threat in ipairs(threats) do
        local distance = (member.root.Position - threat.root.Position).Magnitude
        local bucket = distanceBucket(distance)
        if bucket < nearestBucket
            or (bucket == nearestBucket
                and (not nearest or threat.player.Name < nearest.player.Name)) then
            nearest = threat
            nearestBucket = bucket
        end
    end
    return nearest
end

local function rebuildArenaTargetAssignments(masterPlayer, threats)
    arenaTargetAssignments = {}
    arenaPrimaryFighters = {}

    local healthy = {}
    local lowHealth = {}
    for _, member in ipairs(getArenaMembers(masterPlayer)) do
        if member.healthRatio < CFG.arenaLowHealth then
            table.insert(lowHealth, member)
        else
            table.insert(healthy, member)
        end
    end

    local candidatePairs = {}
    for _, member in ipairs(healthy) do
        for _, threat in ipairs(threats) do
            table.insert(candidatePairs, {
                member = member,
                threat = threat,
                distance = (member.root.Position - threat.root.Position).Magnitude,
            })
        end
    end
    table.sort(candidatePairs, function(left, right)
        local leftBucket = distanceBucket(left.distance)
        local rightBucket = distanceBucket(right.distance)
        if leftBucket ~= rightBucket then
            return leftBucket < rightBucket
        end
        if left.member.player.Name ~= right.member.player.Name then
            return left.member.player.Name < right.member.player.Name
        end
        return left.threat.player.Name < right.threat.player.Name
    end)

    local claimedMembers = {}
    local claimedThreats = {}
    for _, candidate in ipairs(candidatePairs) do
        local memberPlayer = candidate.member.player
        local threatPlayer = candidate.threat.player
        if not claimedMembers[memberPlayer] and not claimedThreats[threatPlayer] then
            claimedMembers[memberPlayer] = true
            claimedThreats[threatPlayer] = true
            arenaTargetAssignments[memberPlayer] = threatPlayer
            arenaPrimaryFighters[threatPlayer] = memberPlayer
        end
    end

    -- 人数多于敌人时，未被选为主战位的成员辅助离自己最近的目标。
    for _, member in ipairs(healthy) do
        if not arenaTargetAssignments[member.player] then
            local nearest = nearestThreatForMember(member, threats)
            arenaTargetAssignments[member.player] = nearest and nearest.player or nil
        end
    end

    -- 残血成员不占用主战名额，只锁定最近威胁用于撤离、保护和近身反击。
    for _, member in ipairs(lowHealth) do
        local nearest = nearestThreatForMember(member, threats)
        arenaTargetAssignments[member.player] = nearest and nearest.player or nil
    end
end

local function getWhitelistSeparation(mRoot)
    local masterPlayer = Players:FindFirstChild(CFG.masterName)
    local separation = Vector3.zero
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player ~= masterPlayer and isWhitelisted(player) then
            local teammateRoot = getAlivePlayerParts(player)
            if teammateRoot then
                local offset = mRoot.Position - teammateRoot.Position
                local distance = offset.Magnitude
                if distance > 0.05 and distance < CFG.whitelistSpacing then
                    separation = separation
                        + offset.Unit * ((CFG.whitelistSpacing - distance) / CFG.whitelistSpacing)
                end
            end
        end
    end
    return separation
end

local function getArenaAssignment(enemyPlayer, eRoot)
    if focusEnabled and not trainingEnabled then return "fighter", nil end
    if not arenaEnabled or trainingEnabled then return "fighter", nil end

    local masterPlayer = Players:FindFirstChild(CFG.masterName)
    local healthy = {}
    local lowHealth = {}
    for _, member in ipairs(getArenaMembers(masterPlayer)) do
        member.enemyDistance = (member.root.Position - eRoot.Position).Magnitude
        if member.healthRatio < CFG.arenaLowHealth then
            table.insert(lowHealth, member)
        else
            table.insert(healthy, member)
        end
    end

    local function nearestFirst(left, right)
        local leftBucket = distanceBucket(left.enemyDistance)
        local rightBucket = distanceBucket(right.enemyDistance)
        if leftBucket ~= rightBucket then
            return leftBucket < rightBucket
        end
        return left.player.Name < right.player.Name
    end
    table.sort(healthy, nearestFirst)
    table.sort(lowHealth, nearestFirst)

    for _, member in ipairs(lowHealth) do
        if member.player == LP then return "retreat", member end
    end

    local fighter = nil
    local primaryPlayer = arenaPrimaryFighters[enemyPlayer]
    for _, member in ipairs(healthy) do
        if member.player == primaryPlayer then
            fighter = member
            break
        end
    end
    if not fighter then
        for _, member in ipairs(healthy) do
            if arenaTargetAssignments[member.player] == enemyPlayer then
                fighter = member
                break
            end
        end
    end
    if fighter and fighter.player == LP then return "fighter", lowHealth[1] end

    local endangered = lowHealth[1]
    if endangered and endangered.enemyDistance < CFG.arenaProtectRange then
        return "protector", endangered
    end
    return "support", endangered
end

local function moveTowardPoint(mRoot, point, separation)
    local delta = Vector3.new(point.X - mRoot.Position.X, 0, point.Z - mRoot.Position.Z)
    local moveDirection = delta.Magnitude > 0.5 and delta.Unit or Vector3.zero
    if separation.Magnitude > 0.05 then
        moveDirection = moveDirection + separation.Unit
    end
    if moveDirection.Magnitude > 0.05 then
        humanoid:Move(moveDirection.Unit, false)
        pcall(function()
            mRoot.CFrame = CFrame.lookAt(mRoot.Position,
                Vector3.new(point.X, mRoot.Position.Y, point.Z))
        end)
    else
        humanoid:Move(Vector3.zero, false)
    end
end

local function moveWhileDefending(mRoot, targetRoot, separation)
    local holdPoint = getRadialPoint(targetRoot, mRoot, CFG.counterApproachDistance)
    if not holdPoint then
        humanoid:Move(Vector3.zero, false)
        return
    end

    local delta = Vector3.new(
        holdPoint.X - mRoot.Position.X,
        0,
        holdPoint.Z - mRoot.Position.Z
    )
    if delta.Magnitude > CFG.counterApproachTolerance then
        moveTowardPoint(mRoot, holdPoint, separation)
    elseif separation.Magnitude > 0.05 then
        humanoid:Move(separation.Unit, false)
    else
        humanoid:Move(Vector3.zero, false)
    end
end

local function tryRagdollAirBlitz(distance)
    if ultimateActive or distance > 50 or blitzJumpPending or not enemyIsRagdoll then
        return false
    end
    if not isSkillReady("blitz") or not humanoid then return false end

    blitzJumpPending = true
    task.spawn(function()
        local startedAt = tick()
        pcall(function()
            humanoid.Jump = true
            humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end)

        while tick() - startedAt < 0.9 do
            if mode ~= "combat" or isStandby or not trackEnabled
                or ultimateActive or not enemyIsRagdoll or not currentEnemyChar then
                break
            end

            if humanoid.FloorMaterial == Enum.Material.Air then
                performSkill("blitz")
                break
            end
            RunService.Heartbeat:Wait()
        end
        blitzJumpPending = false
    end)
    return true
end

-- 动画检测辅助
local function animTrackHasId(track, id)
    local animation = track and track.Animation
    local animationId = animation and animation.AnimationId
    return animationId ~= nil and tostring(animationId):find(id, 1, true) ~= nil
end

local function animTrackHasAnyId(track, ids)
    for _, id in ipairs(ids) do
        if animTrackHasId(track, id) then return true end
    end
    return false
end

local function animHasId(animator, id)
    local ok, tracks = pcall(function() return animator:GetPlayingAnimationTracks() end)
    if not ok then return false end
    for _, t in ipairs(tracks) do
        if animTrackHasId(t, id) then return true end
    end
    return false
end


local function animHasAnyId(animator, ids)
    local ok, tracks = pcall(function() return animator:GetPlayingAnimationTracks() end)
    if not ok then return false end
    for _, track in ipairs(tracks) do
        if animTrackHasAnyId(track, ids) then return true end
    end
    return false
end

local function getActiveGuardRadius()
    if ultimateActive then
        return CFG.guardRadiusBoosted
    end
    return CFG.guardRadius
end

local function deactivateUltimate(reason)
    if not ultimateActive and not ultimateTrying then return end
    ultimateActive = false
    ultimateTrying = false
    ultimateSkillsReady = false
    ultimateAnimDetected = false
    ultimateActivatedAt = 0
    lastSkillUsed = ""
    if reason then
        setStatus(reason, Color3.fromRGB(100, 200, 255))
    end
end

local function activateUltimate(now)
    cancelCounterAttack()
    stopAttacking()
    blitzJumpPending = false
    ultimateActive = true
    ultimateTrying = false
    ultimateSkillsReady = false
    ultimateAnimDetected = true
    ultimateActivatedAt = 0
    lastUltimateTry = now
    lastSkillUsed = ""
    setStatus("状态: ⚡ 大招开启动画中", Color3.fromRGB(255, 210, 80))
end

local function finishUltimateTransform(now)
    ultimateSkillsReady = true
    ultimateActivatedAt = now
    skillLastUsed.thunderKick = 0
    skillLastUsed.speedblitz = 0
    skillLastUsed.flamewave = 0
    skillLastUsed.incinerate = 0
    setStatus("状态: ⚡ 大招技能已就绪", Color3.fromRGB(255, 210, 80))
end

local function tryActivateUltimate(now)
    if mode ~= "combat" or not trackEnabled or isStandby or ultimateActive then
        return false
    end
    if ultimateTrying then
        if now - lastUltimateTry > CFG.ultimateTryWindow then
            ultimateTrying = false
            ultimateAnimDetected = false
        else
            return false
        end
    end
    if now - lastUltimateTry < CFG.ultimateTryInterval then return false end

    lastUltimateTry = now
    ultimateTrying = true
    ultimateAnimDetected = false
    sendKey(Enum.KeyCode.G, true)
    task.delay(0.12, function()
        sendKey(Enum.KeyCode.G, false)
    end)
    setStatus("状态: ⚡ 尝试开启大招", Color3.fromRGB(255, 200, 90))
    return true
end

local function updateUltimateState(now)
    local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
    local found = animator and animHasId(animator, CFG.ultimateAnimId) or false

    if ultimateActive then
        -- 成功动画结束后才切换技能，并从此刻开始计算大招持续时间。
        if not ultimateSkillsReady then
            if found then
                ultimateAnimDetected = true
            elseif ultimateAnimDetected then
                finishUltimateTransform(now)
            end
            return
        end

        if now - ultimateActivatedAt >= CFG.ultimateDuration then
            deactivateUltimate("状态: ⚡ 大招结束，回普通")
        end
        return
    end

    if not ultimateTrying then return end
    if now - lastUltimateTry > CFG.ultimateTryWindow then
        ultimateTrying = false
        ultimateAnimDetected = false
        return
    end

    if found and not ultimateAnimDetected then
        activateUltimate(now)
    elseif not found then
        ultimateAnimDetected = false
    end
end

local function useCombatSkill(distance)
    if ultimateActive then
        if not ultimateSkillsReady then return end
        -- 大招技能：近身踢击、全距离冲刺踢、40米外双火炮。
        if distance <= CFG.skills.thunderKick.range and performSkill("thunderKick") then
            return
        end
        if distance > 40 then
            if performSkill("flamewave") then return end
            if performSkill("incinerate") then return end
        end
        performSkill("speedblitz")
        return
    end

    -- 近身连打；中距离突进；极限距离冲刺轰击；50米外使用远程火炮。
    if distance <= CFG.skills.machineGun.range and performSkill("machineGun") then
        return
    end
    if distance >= CFG.skills.ignition.minRange
        and distance <= CFG.skills.ignition.range
        and performSkill("ignition") then
        return
    end
    if distance >= CFG.skills.jetDive.minRange
        and distance <= CFG.skills.jetDive.range
        and performSkill("jetDive") then
        return
    end
    if distance > 50 then
        performSkill("blitz")
    end
end


local function getBlitzAimPoint(eRoot)
    local velocity = eRoot.AssemblyLinearVelocity
    local horizontalVelocity = Vector3.new(velocity.X, 0, velocity.Z)
    if horizontalVelocity.Magnitude < CFG.blitzMoveThreshold then
        return eRoot.Position, false
    end

    return eRoot.Position + horizontalVelocity.Unit * CFG.blitzLeadDistance, true
end

local function updateBlitzPreAim(mRoot, eRoot)
    local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
    if not animator or not animHasId(animator, CFG.blitzAnimId) then
        return false
    end

    local aimPoint = getBlitzAimPoint(eRoot)
    if humanoid.FloorMaterial == Enum.Material.Air then
        aimPoint = aimPoint - Vector3.new(0, CFG.blitzAirAimDrop, 0)
    end
    local flatAim = Vector3.new(aimPoint.X, mRoot.Position.Y, aimPoint.Z)
    pcall(function()
        if (flatAim - mRoot.Position).Magnitude > 0.05 then
            mRoot.CFrame = CFrame.lookAt(mRoot.Position, flatAim)
        end
        if (aimPoint - camera.CFrame.Position).Magnitude > 0.05 then
            camera.CFrame = CFrame.lookAt(camera.CFrame.Position, aimPoint)
        end
    end)
    return true
end

-- ══════════════════════════════════════
--  威胁检测：扫描主人附近的敌人，并为白名单成员就近分配目标
-- ══════════════════════════════════════
local function findNearestThreat()
    local masterPlayer = Players:FindFirstChild(CFG.masterName)
    if not masterPlayer or not masterPlayer.Character then return nil end
    local masterRoot = masterPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not masterRoot then return nil end

    arenaTargetAssignments = {}
    arenaPrimaryFighters = {}

    if trainingEnabled then
        local masterHumanoid = masterPlayer.Character:FindFirstChild("Humanoid")
        if masterHumanoid and masterHumanoid.Health > 0 then
            return masterPlayer
        end
        return nil
    end

    local threats = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p ~= masterPlayer and not isWhitelisted(p) then
            local root, enemyHumanoid = getAlivePlayerParts(p)
            if root and enemyHumanoid then
                local masterDistance = (root.Position - masterRoot.Position).Magnitude
                if masterDistance <= getActiveGuardRadius() then
                    table.insert(threats, {
                        player = p,
                        root = root,
                        masterDistance = masterDistance,
                    })
                end
            end
        end
    end
    table.sort(threats, function(left, right)
        return left.player.Name < right.player.Name
    end)
    if #threats == 0 then return nil end

    -- 集中模式下，所有白名单实例都使用主人作为距离基准，锁定同一个最近目标。
    if focusEnabled then
        table.sort(threats, function(left, right)
            if left.masterDistance ~= right.masterDistance then
                return left.masterDistance < right.masterDistance
            end
            return left.player.Name < right.player.Name
        end)
        return threats[1].player
    end

    if arenaEnabled then
        rebuildArenaTargetAssignments(masterPlayer, threats)
        return arenaTargetAssignments[LP]
    end

    local localRoot = getAlivePlayerParts(LP)
    local referencePosition = localRoot and localRoot.Position or masterRoot.Position
    local nearest, nearestBucket = nil, math.huge
    for _, threat in ipairs(threats) do
        local distance = (threat.root.Position - referencePosition).Magnitude
        local bucket = distanceBucket(distance)
        if bucket < nearestBucket
            or (bucket == nearestBucket
                and (not nearest or threat.player.Name < nearest.player.Name)) then
            nearest = threat
            nearestBucket = bucket
        end
    end
    return nearest and nearest.player or nil
end

-- 连携技始终以主人为距离基准，不受集中/擂台/练招目标分配影响。
local function findNearestThreatToMaster()
    local masterPlayer = Players:FindFirstChild(CFG.masterName)
    local masterRoot = masterPlayer
        and masterPlayer.Character
        and masterPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not masterRoot then return nil end

    local nearest, nearestDistance = nil, math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player ~= masterPlayer and not isWhitelisted(player) then
            local root, playerHumanoid = getAlivePlayerParts(player)
            if root and playerHumanoid then
                local distance = (root.Position - masterRoot.Position).Magnitude
                if distance <= getActiveGuardRadius()
                    and (distance < nearestDistance
                        or (distance == nearestDistance
                            and (not nearest or player.Name < nearest.Name))) then
                    nearest = player
                    nearestDistance = distance
                end
            end
        end
    end
    return nearest
end

local function triggerComboAttack(targetPlayer)
    if not comboEnabled or comboExecuting or LP.Name ~= CFG.comboOwnerName then return false end

    local targetCharacter = targetPlayer and targetPlayer.Character
    local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
    local targetHumanoid = targetCharacter and targetCharacter:FindFirstChild("Humanoid")
    local localRoot = character and character:FindFirstChild("HumanoidRootPart")
    if not targetRoot or not targetHumanoid or targetHumanoid.Health <= 0 or not localRoot then
        return false
    end

    comboExecuting = true
    comboWindowUntil = 0
    comboEnemyAnimDetected = true
    comboTarget = targetPlayer
    comboGeneration = comboGeneration + 1
    local generation = comboGeneration

    task.spawn(function()
        local comboSideDashing = false
        local comboHoldingDistance = false

        local function releaseComboControls()
            pcall(function()
                RunService:UnbindFromRenderStep(COMBO_CAMERA_BIND_KEY)
            end)
            sendKey(Enum.KeyCode.Q, false)
            sendKey(Enum.KeyCode.D, false)
        end

        local function comboStillValid()
            return runtime.alive
                and generation == comboGeneration
                and comboEnabled
                and comboExecuting
                and localRoot
                and localRoot.Parent
                and targetRoot
                and targetRoot.Parent
                and targetHumanoid
                and targetHumanoid.Parent
                and targetHumanoid.Health > 0
        end

        local function returnComboToStandby()
            releaseComboControls()
            if not runtime.alive or generation ~= comboGeneration or not comboEnabled then
                return false
            end

            -- 目标死亡或离场也必须持续回地下，不能把返程依赖于目标存活。
            local returnStartedAt = tick()
            while runtime.alive and generation == comboGeneration and comboEnabled
                and tick() - returnStartedAt < CFG.comboReturnEnforceTime do
                if not isStandby or not standbyAnchor then
                    enterStandby()
                end
                local root = character and character:FindFirstChild("HumanoidRootPart")
                if root and humanoid then
                    pcall(function()
                        if next(standbySavedParts) == nil then
                            applyStandbyStealth(true)
                        end
                        lockStandbyPose(root)
                    end)
                end
                RunService.Heartbeat:Wait()
            end

            if generation ~= comboGeneration or not comboEnabled then return false end
            comboExecuting = false
            comboTarget = nil
            setStatus("状态: 连携完成，地下隐身待机", Color3.fromRGB(180, 130, 255))
            return true
        end

        local behind = targetRoot.Position - targetRoot.CFrame.LookVector * CFG.comboBehindDistance
        local destination = CFrame.lookAt(behind, targetRoot.Position)
        exitStandby(destination)

        if not comboStillValid() then
            returnComboToStandby()
            return
        end
        pcall(function()
            RunService:UnbindFromRenderStep(COMBO_CAMERA_BIND_KEY)
        end)
        RunService:BindToRenderStep(COMBO_CAMERA_BIND_KEY, Enum.RenderPriority.Camera.Value + 1, function()
            if generation ~= comboGeneration or not comboEnabled or not comboExecuting
                or not targetRoot or not targetRoot.Parent or targetHumanoid.Health <= 0 then
                RunService:UnbindFromRenderStep(COMBO_CAMERA_BIND_KEY)
                return
            end
            pcall(function()
                if comboHoldingDistance then
                    local holdPosition = targetRoot.Position
                        - targetRoot.CFrame.LookVector * CFG.comboFirstAttackHoldDistance
                    humanoid:Move(Vector3.zero, false)
                    localRoot.AssemblyLinearVelocity = Vector3.zero
                    localRoot.AssemblyAngularVelocity = Vector3.zero
                    localRoot.CFrame = CFrame.lookAt(holdPosition, targetRoot.Position)
                elseif comboSideDashing then
                    -- 让角色右方向始终指向目标：D+Q 仍是右侧闪，但位移会向目标收近。
                    local flatToTarget = Vector3.new(
                        targetRoot.Position.X - localRoot.Position.X,
                        0,
                        targetRoot.Position.Z - localRoot.Position.Z
                    )
                    if flatToTarget.Magnitude > 0.05 then
                        local dashLook = Vector3.yAxis:Cross(flatToTarget.Unit)
                        localRoot.CFrame = CFrame.lookAt(
                            localRoot.Position,
                            localRoot.Position + dashLook
                        )
                    end
                elseif not comboSideDashing then
                    humanoid:Move(Vector3.zero, false)
                    localRoot.AssemblyLinearVelocity = Vector3.zero
                    localRoot.AssemblyAngularVelocity = Vector3.zero
                    localRoot.CFrame = CFrame.lookAt(localRoot.Position, targetRoot.Position)
                end
                camera.CFrame = CFrame.lookAt(camera.CFrame.Position, targetRoot.Position)
            end)
        end)

        task.wait(CFG.comboPreAttackDelay)
        if not comboStillValid() then
            returnComboToStandby()
            return
        end

        fireServer({ { Goal = "LeftClick", Mobile = true } })
        task.wait(CFG.comboAttackPressTime)
        fireServer({ { Goal = "LeftClickRelease", Mobile = true } })

        comboHoldingDistance = true
        task.wait(CFG.comboSecondAttackDelay)
        comboHoldingDistance = false
        if not comboStillValid() then
            returnComboToStandby()
            return
        end

        -- 从实时的8米保持位发起纯右侧闪；渲染回调会把角色右方向动态对准目标。
        local secondBehind = targetRoot.Position
            - targetRoot.CFrame.LookVector * CFG.comboFirstAttackHoldDistance
        local flatToTarget = Vector3.new(
            targetRoot.Position.X - secondBehind.X,
            0,
            targetRoot.Position.Z - secondBehind.Z
        )
        if flatToTarget.Magnitude > 0.05 then
            local dashLook = Vector3.yAxis:Cross(flatToTarget.Unit)
            localRoot.CFrame = CFrame.lookAt(secondBehind, secondBehind + dashLook)
        else
            localRoot.CFrame = CFrame.lookAt(secondBehind, targetRoot.Position)
        end
        localRoot.AssemblyLinearVelocity = Vector3.zero
        localRoot.AssemblyAngularVelocity = Vector3.zero

        comboSideDashing = true
        performComboSideDash()
        comboSideDashing = false
        if not comboStillValid() then
            returnComboToStandby()
            return
        end

        -- 保留侧闪形成的右侧方位，并把结束点收敛到近身攻击距离。
        local horizontalOffset = Vector3.new(
            localRoot.Position.X - targetRoot.Position.X,
            0,
            localRoot.Position.Z - targetRoot.Position.Z
        )
        if horizontalOffset.Magnitude < 0.05 then
            horizontalOffset = -Vector3.new(
                targetRoot.CFrame.LookVector.X,
                0,
                targetRoot.CFrame.LookVector.Z
            )
        end
        local closePosition = targetRoot.Position
            + horizontalOffset.Unit * CFG.comboSecondAttackCloseDistance
        localRoot.CFrame = CFrame.lookAt(closePosition, targetRoot.Position)

        fireServer({ { Goal = "LeftClick", Mobile = true } })
        task.wait(CFG.comboAttackPressTime)
        fireServer({ { Goal = "LeftClickRelease", Mobile = true } })

        task.wait(math.max(0, CFG.comboReturnDelay - CFG.comboAttackPressTime))
        returnComboToStandby()
    end)
    return true
end

-- ══════════════════════════════════════
--  进入守卫模式
-- ══════════════════════════════════════
local function enterGuardMode()
    if mode == "guard" then
        clearEnemyBackstep()
        return
    end
    mode         = "guard"
    clearEnemyBackstep()
    currentEnemy     = nil
    currentEnemyChar = nil
    enemyIsRagdoll   = false
    arenaRole        = "fighter"
    blitzJumpPending = false
    cancelCounterAttack()
    stopAttacking()

    -- 清除敌方相关连接（保留master和自身连接）
    if CONN.enemyDied    then CONN.enemyDied:Disconnect();    CONN.enemyDied    = nil end
    if CONN.enemyRagdoll then CONN.enemyRagdoll:Disconnect(); CONN.enemyRagdoll = nil end
    if CONN.enemyState   then CONN.enemyState:Disconnect();   CONN.enemyState   = nil end
    if CONN.enemyLeft    then CONN.enemyLeft:Disconnect();    CONN.enemyLeft    = nil end
    if CONN.counterAnim  then CONN.counterAnim:Disconnect();  CONN.counterAnim  = nil end
    if CONN.enemyBackstepPlayed then
        CONN.enemyBackstepPlayed:Disconnect()
        CONN.enemyBackstepPlayed = nil
    end
end

-- ══════════════════════════════════════
--  进入战斗模式（锁定新敌方）
-- ══════════════════════════════════════
local function enterCombatMode(enemyPlayer)
    if currentEnemy == enemyPlayer and mode == "combat" then return end
    clearEnemyBackstep()
    mode             = "combat"
    currentEnemy     = enemyPlayer
    currentEnemyChar = enemyPlayer.Character
    enemyIsRagdoll   = false
    cancelCounterAttack()
    stopAttacking()

    -- 清旧敌方连接
    if CONN.enemyDied    then CONN.enemyDied:Disconnect()    end
    if CONN.enemyRagdoll then CONN.enemyRagdoll:Disconnect() end
    if CONN.enemyState   then CONN.enemyState:Disconnect()   end
    if CONN.enemyLeft    then CONN.enemyLeft:Disconnect()      end
    if CONN.counterAnim  then CONN.counterAnim:Disconnect()    end
    if CONN.enemyBackstepPlayed then CONN.enemyBackstepPlayed:Disconnect() end

    local eHum = currentEnemyChar and currentEnemyChar:FindFirstChild("Humanoid")
    if not eHum then return end

    -- 敌方死亡 → 回守卫
    CONN.enemyDied = eHum.Died:Connect(function()
        enterGuardMode()
        setStatus("状态: 🛡 目标击败，回守卫", Color3.fromRGB(100, 200, 255))
    end)

    -- 敌方离开（角色被移除）
    CONN.enemyLeft = currentEnemy.CharacterRemoving:Connect(function()
        enterGuardMode()
    end)

    -- 敌方布娃娃检测
    CONN.enemyRagdoll = eHum:GetPropertyChangedSignal("PlatformStand"):Connect(function()
        enemyIsRagdoll = eHum.PlatformStand
        if enemyIsRagdoll then
            stopAttacking()
            timedGuardActive = false
            timedGuardUntil = 0
            alternateTimedGuardActive = false
            alternateTimedGuardStartDetected = false
            alternateTimedGuardUntil = 0
            quickGuardActive = false
            quickGuardStartDetected = false
            quickGuardUntil = 0
            setSustainedGuard(chainedGuardActive)
        end
    end)
    CONN.enemyState = eHum.StateChanged:Connect(function(_, new)
        if new == Enum.HumanoidStateType.Physics then
            enemyIsRagdoll = true
            stopAttacking()
            timedGuardActive = false
            timedGuardUntil = 0
            alternateTimedGuardActive = false
            alternateTimedGuardStartDetected = false
            alternateTimedGuardUntil = 0
            quickGuardActive = false
            quickGuardStartDetected = false
            quickGuardUntil = 0
            setSustainedGuard(chainedGuardActive)
        elseif new == Enum.HumanoidStateType.GettingUp
            or new == Enum.HumanoidStateType.Running
            or new == Enum.HumanoidStateType.Landed
            or new == Enum.HumanoidStateType.RunningNoPhysics then
            enemyIsRagdoll = false
        end
    end)

    -- Humanoid provides the same track API when Animator replication is late.
    local eAnimator = eHum:FindFirstChildOfClass("Animator") or eHum
    if eAnimator then
        CONN.enemyBackstepPlayed = eAnimator.AnimationPlayed:Connect(function(track)
            if mode == "combat" and trackEnabled and not isStandby
                and currentEnemy == enemyPlayer then
                if animTrackHasId(track, CFG.enemyBackstepAnimId) then
                    activateEnemyBackstep(enemyPlayer)
                    enemyBackstepAnimPlaying = true
                end
                if animTrackHasId(track, CFG.timedGuardAnimId) then
                    startTimedGuard()
                end
                if animTrackHasId(track, CFG.alternateTimedGuardAnimId) then
                    startAlternateTimedGuard()
                end
                if animTrackHasId(track, CFG.quickGuardAnimId) then
                    startQuickGuard()
                end
            end
        end)

        CONN.counterAnim = RunService.Heartbeat:Connect(function()
            if mode ~= "combat" or not trackEnabled or isStandby or not currentEnemyChar
                or (ultimateActive and not ultimateSkillsReady) then
                clearEnemyBackstep()
                counterAnimDetected = false
                counterTriggeredForAnimation = false
                chainedGuardActive = false
                chainedGuardEndDetected = false
                chainedGuardCounterReady = false
                timedGuardActive = false
                timedGuardStartDetected = false
                timedGuardUntil = 0
                alternateTimedGuardActive = false
                alternateTimedGuardStartDetected = false
                alternateTimedGuardUntil = 0
                quickGuardActive = false
                quickGuardStartDetected = false
                quickGuardUntil = 0
                setSustainedGuard(false)
                reactiveDashAnimDetected = false
                reactiveDashTriggered = false
                return
            end

            local mRoot = character and character:FindFirstChild("HumanoidRootPart")
            local eRoot = currentEnemyChar:FindFirstChild("HumanoidRootPart")
            if not mRoot or not eRoot then
                clearEnemyBackstep()
                counterAnimDetected = false
                counterTriggeredForAnimation = false
                chainedGuardActive = false
                chainedGuardEndDetected = false
                chainedGuardCounterReady = false
                timedGuardActive = false
                timedGuardStartDetected = false
                timedGuardUntil = 0
                alternateTimedGuardActive = false
                alternateTimedGuardStartDetected = false
                alternateTimedGuardUntil = 0
                quickGuardActive = false
                quickGuardStartDetected = false
                quickGuardUntil = 0
                setSustainedGuard(false)
                reactiveDashAnimDetected = false
                reactiveDashTriggered = false
                return
            end

            -- AnimationPlayed 负责即时响应；轮询负责补偿监听绑定前已经开始的动画。
            local enemyBackstepFound = animHasId(eAnimator, CFG.enemyBackstepAnimId)
            if enemyBackstepFound then
                if not enemyBackstepAnimPlaying then
                    activateEnemyBackstep(enemyPlayer)
                else
                    enemyBackstepUntil = tick() + CFG.enemyBackstepWindow
                end
            end
            enemyBackstepAnimPlaying = enemyBackstepFound

            -- Fallback for a root that appears after the animation event.
            -- The full defense track remains protected until it expires.
            if isEnemyBackstepActive(eRoot)
                and not enemyBackstepCloseMode
                and (eRoot.Position - mRoot.Position).Magnitude <= CFG.enemyBackstepTriggerRange then
                enemyBackstepCloseMode = true
                enemyBackstepAdvanceAttack = isOutsideEnemyFront(mRoot, eRoot, false)
                isAttacking = false
                fireServer({ { Goal = "LeftClickRelease", Mobile = true } })
            end

            local chainedStartFound = animHasId(eAnimator, CFG.chainedGuardStartAnimId)
            local chainedEndFound = false
            for _, animationId in ipairs(CFG.chainedGuardEndAnimIds) do
                if animHasId(eAnimator, animationId) then
                    chainedEndFound = true
                    break
                end
            end

            if chainedStartFound and not chainedGuardActive then
                chainedGuardActive = true
                chainedGuardEndDetected = false
                chainedGuardCounterReady = false
                stopAttacking()
            end
            if chainedGuardActive then
                if chainedEndFound then
                    chainedGuardEndDetected = true
                elseif chainedGuardEndDetected then
                    chainedGuardActive = false
                    chainedGuardEndDetected = false
                    chainedGuardCounterReady = true
                end
            end

            local timedGuardStartFound = animHasId(eAnimator, CFG.timedGuardAnimId)
            if timedGuardStartFound then
                startTimedGuard()
            else
                timedGuardStartDetected = false
            end

            local timedGuardEndFound = animHasId(eAnimator, CFG.timedGuardEndAnimId)
            if timedGuardActive and (timedGuardEndFound or enemyIsRagdoll
                or tick() >= timedGuardUntil) then
                timedGuardActive = false
                timedGuardUntil = 0
            end

            local alternateTimedGuardStartFound = animHasId(eAnimator, CFG.alternateTimedGuardAnimId)
            if alternateTimedGuardStartFound then
                startAlternateTimedGuard()
            else
                alternateTimedGuardStartDetected = false
            end

            local alternateTimedGuardEndFound = animHasId(eAnimator, CFG.alternateTimedGuardEndAnimId)
            if alternateTimedGuardActive and (alternateTimedGuardEndFound or enemyIsRagdoll
                or tick() >= alternateTimedGuardUntil) then
                alternateTimedGuardActive = false
                alternateTimedGuardUntil = 0
            end

            local quickGuardStartFound = animHasId(eAnimator, CFG.quickGuardAnimId)
            if quickGuardStartFound then
                startQuickGuard()
            else
                quickGuardStartDetected = false
            end

            if quickGuardActive and (enemyIsRagdoll or tick() >= quickGuardUntil) then
                quickGuardActive = false
                quickGuardUntil = 0
            end

            setSustainedGuard(chainedGuardActive or timedGuardActive
                or alternateTimedGuardActive or quickGuardActive)
            if chainedGuardCounterReady and not sustainedGuardActive then
                chainedGuardCounterReady = false
                triggerCounterAttack()
            end

            local found = animHasId(eAnimator, CFG.counterAnimId)
            if found then
                local distance = (eRoot.Position - mRoot.Position).Magnitude
                if not counterAnimDetected and distance <= CFG.counterDetectRange then
                    counterAnimDetected = true
                    stopAttacking()
                end
                if counterAnimDetected
                    and distance <= CFG.counterRange
                    and not counterTriggeredForAnimation then
                    counterTriggeredForAnimation = true
                    triggerCounterAttack()
                end
            else
                counterAnimDetected = false
                counterTriggeredForAnimation = false
            end

            local reactiveFound = animHasId(eAnimator, CFG.reactiveDashAnimId)
            if reactiveFound then
                reactiveDashAnimDetected = true
                if not reactiveDashTriggered and triggerReactiveDash() then
                    reactiveDashTriggered = true
                end
            else
                reactiveDashAnimDetected = false
                reactiveDashTriggered = false
            end
        end)
    end

end

-- ══════════════════════════════════════
--  主循环
-- ══════════════════════════════════════
local guardAndCombat  -- 前向声明

guardAndCombat = function()
    if not runtime.alive then return end
    disconnectAll()
    stopAttacking()
    comboGeneration = comboGeneration + 1
    comboExecuting = false
    comboWindowUntil = 0
    comboMasterAnimDetected = false
    comboEnemyAnimDetected = false
    comboTarget = nil
    comboObservedEnemy = nil
    clearEnemyBackstep()
    mode             = "guard"
    currentEnemy     = nil
    currentEnemyChar = nil
    enemyIsRagdoll   = false
    arenaRole        = "fighter"
    blitzJumpPending = false
    cancelCounterAttack()
    -- 待机状态重置（重生后解除待机）
    if isStandby then
        exitStandby()
    end
    masterAnimCount    = 0
    masterAnimTimer    = 0
    masterAnimDetected = false

    if comboEnabled and LP.Name == CFG.comboOwnerName then
        task.defer(function()
            if comboEnabled and trackEnabled then
                enterStandby()
            end
        end)
    end

    local masterPlayer = Players:FindFirstChild(CFG.masterName)
    if not masterPlayer then
        warn("[AutoBattle] 未找到主人: " .. CFG.masterName)
        setStatus("状态: ❌ 主人未找到", Color3.fromRGB(255, 80, 80))
        return
    end

    local masterChar     = masterPlayer.Character or masterPlayer.CharacterAdded:Wait()
    local masterHumanoid = masterChar:WaitForChild("Humanoid")

    local localAnimator = humanoid and (humanoid:FindFirstChildOfClass("Animator") or humanoid)
    if localAnimator then
        CONN.selfBackFanPlayed = localAnimator.AnimationPlayed:Connect(function(track)
            if runtime.alive and animTrackHasAnyId(track, CFG.selfBackFanAnimIds) then
                triggerSelfBackFanTeleport()
            end
        end)
    end

    -- 主人死亡（只清战斗相关连接，保留masterRespawn）
    CONN.masterDied = masterHumanoid.Died:Connect(function()
        comboGeneration = comboGeneration + 1
        comboExecuting = false
        comboWindowUntil = 0
        comboMasterAnimDetected = false
        comboEnemyAnimDetected = false
        comboTarget = nil
        comboObservedEnemy = nil
        if comboEnabled and not isStandby then enterStandby() end
        -- 单独清除除 masterRespawn 以外的所有连接
        for k, c in pairs(CONN) do
            if k ~= "masterRespawn" then
                if c then pcall(function() c:Disconnect() end) end
                CONN[k] = nil
            end
        end
        stopAttacking()
        setStatus("状态: 💀 主人死亡，等待复活...", Color3.fromRGB(200, 80, 80))
    end)

    -- 主人重生
    CONN.masterRespawn = masterPlayer.CharacterAdded:Connect(function(newChar)
        masterChar = newChar
        task.spawn(function()
            task.wait(0.5)
            guardAndCombat()
        end)
    end)

    -- 主人动画三击检测：
    --   2秒内连续 2次 → 切换待机/恢复
    --   2秒内连续 3次 → 强制召唤自己到主人身边（无论是否待机）
    local mAnimator = masterHumanoid:FindFirstChildOfClass("Animator")
    if mAnimator then
        local function bindComboEnemyAnimation(nearest)
            if CONN.comboEnemyPlayed then
                CONN.comboEnemyPlayed:Disconnect()
                CONN.comboEnemyPlayed = nil
            end

            comboObservedEnemy = nearest
            comboEnemyAnimDetected = false
            local enemyHumanoid = nearest
                and nearest.Character
                and nearest.Character:FindFirstChild("Humanoid")
            local enemyAnimator = enemyHumanoid
                and enemyHumanoid:FindFirstChildOfClass("Animator")
            if not enemyAnimator then return nil end

            CONN.comboEnemyPlayed = enemyAnimator.AnimationPlayed:Connect(function(track)
                if not runtime.alive or not comboEnabled or comboExecuting then return end
                if tick() > comboWindowUntil then return end
                if nearest ~= findNearestThreatToMaster() then return end
                if animTrackHasAnyId(track, CFG.comboEnemyAnimIds) then
                    comboEnemyAnimDetected = true
                    triggerComboAttack(nearest)
                end
            end)
            return enemyAnimator
        end

        local function openComboWindow()
            if not comboEnabled or comboExecuting then return end
            comboMasterAnimDetected = true
            comboWindowUntil = tick() + CFG.comboWindow
            local nearest = findNearestThreatToMaster()
            local enemyAnimator = bindComboEnemyAnimation(nearest)
            setStatus("状态: 连携窗口开启，等待最近敌人动作", Color3.fromRGB(255, 190, 80))

            -- 两个动画近乎同时开始时，AnimationPlayed 可能早于监听绑定；立即查询一次补偿。
            if nearest and enemyAnimator and animHasAnyId(enemyAnimator, CFG.comboEnemyAnimIds) then
                comboEnemyAnimDetected = true
                triggerComboAttack(nearest)
            end
        end

        CONN.comboMasterPlayed = mAnimator.AnimationPlayed:Connect(function(track)
            if runtime.alive and comboEnabled and animTrackHasId(track, CFG.comboMasterAnimId) then
                openComboWindow()
            end
        end)

        CONN.masterAnim = RunService.Heartbeat:Connect(function()
            if comboEnabled then
                local now = tick()
                local comboMasterFound = animHasId(mAnimator, CFG.comboMasterAnimId)
                if comboMasterFound and not comboMasterAnimDetected then
                    openComboWindow()
                elseif not comboMasterFound then
                    comboMasterAnimDetected = false
                end

                if not comboExecuting and now <= comboWindowUntil then
                    local nearest = findNearestThreatToMaster()
                    local enemyHumanoid = nearest
                        and nearest.Character
                        and nearest.Character:FindFirstChild("Humanoid")
                    local enemyAnimator = enemyHumanoid
                        and enemyHumanoid:FindFirstChildOfClass("Animator")
                    local comboEnemyFound = enemyAnimator
                        and animHasAnyId(enemyAnimator, CFG.comboEnemyAnimIds)
                        or false

                    if nearest ~= comboObservedEnemy then
                        enemyAnimator = bindComboEnemyAnimation(nearest)
                        comboEnemyFound = enemyAnimator
                            and animHasAnyId(enemyAnimator, CFG.comboEnemyAnimIds)
                            or false
                        if comboEnemyFound then
                            comboEnemyAnimDetected = true
                            triggerComboAttack(nearest)
                        end
                    elseif comboEnemyFound and not comboEnemyAnimDetected then
                        comboEnemyAnimDetected = true
                        triggerComboAttack(nearest)
                    elseif not comboEnemyFound then
                        comboEnemyAnimDetected = false
                    end
                elseif now > comboWindowUntil then
                    comboWindowUntil = 0
                    comboEnemyAnimDetected = false
                    comboObservedEnemy = nil
                    if CONN.comboEnemyPlayed then
                        CONN.comboEnemyPlayed:Disconnect()
                        CONN.comboEnemyPlayed = nil
                    end
                end
                return
            end

            local found = animHasId(mAnimator, CFG.teleportAnimId)
            if found and not masterAnimDetected then
                masterAnimDetected = true
                local now = tick()

                -- 超出时间窗口，重置为第一次
                if masterAnimCount > 0 and (now - masterAnimTimer) > masterAnimWindow then
                    masterAnimCount = 0
                end

                masterAnimCount = masterAnimCount + 1
                if masterAnimCount == 1 then
                    masterAnimTimer = now  -- 开始计时
                end

                if masterAnimCount == 3 then
                    -- ── 第三次：强制召唤到主人身边 ──
                    masterAnimCount = 0
                    masterAnimTimer = 0
                    local mRoot      = character and character:FindFirstChild("HumanoidRootPart")
                    local masterRoot = masterChar and masterChar:FindFirstChild("HumanoidRootPart")
                    if mRoot and masterRoot then
                        -- 退出待机状态
                        if isStandby then
                            exitStandby()
                        end
                        local offset = masterRoot.CFrame.RightVector * 2
                        pcall(function()
                            mRoot.CFrame = CFrame.new(masterRoot.Position + offset)
                        end)
                        setStatus("状态: 🌟 已召唤回主人身边", Color3.fromRGB(255, 220, 60))
                    end

                elseif masterAnimCount == 2 then
                    -- ── 第二次：切换待机/恢复 ──
                    if not isStandby then
                        -- 进入待机：沉入地下隐身点并持续锁定。
                        if enterStandby() then
                            setStatus("状态: 🌙 地下隐身待机中", Color3.fromRGB(180, 130, 255))
                        end
                    else
                        -- 退出待机：解锁并传送回主人身边
                        local mRoot      = character and character:FindFirstChild("HumanoidRootPart")
                        local masterRoot = masterChar and masterChar:FindFirstChild("HumanoidRootPart")
                        if mRoot and masterRoot then
                            local offset = masterRoot.CFrame.RightVector * 2
                            exitStandby(CFrame.new(masterRoot.Position + offset))
                        else
                            exitStandby()
                        end
                        setStatus("状态: 🛡 恢复守卫", Color3.fromRGB(100, 200, 255))
                    end
                end

            elseif not found then
                masterAnimDetected = false
            end
        end)
    else
        warn("[AutoBattle] 主人无Animator，待机手势检测跳过")
    end

    -- 主循环
    CONN.main = RunService.RenderStepped:Connect(function()
        if not trackEnabled then return end

        local mRoot = character and character:FindFirstChild("HumanoidRootPart")
        if not mRoot then return end

        if comboEnabled then
            stopAttacking()
            if not comboExecuting then
                if not isStandby then enterStandby() end
                pcall(function()
                    if next(standbySavedParts) == nil then
                        applyStandbyStealth(true)
                    end
                    lockStandbyPose(mRoot)
                end)
            end
            return
        end

        -- ── 地下隐身待机：每帧锁回地下锚点，并维持无碰撞/隐藏 ──
        if isStandby then
            stopAttacking()
            pcall(function()
                -- 若角色部件被游戏重建，补一次隐身/无碰撞
                if next(standbySavedParts) == nil then
                    applyStandbyStealth(true)
                end
                lockStandbyPose(mRoot)
            end)
            return
        end

        local now = tick()
        updateUltimateState(now)

        -- ── 定期扫描威胁 ──
        if now - lastThreatCheck >= CFG.threatCheckRate then
            lastThreatCheck = now
            local threat = findNearestThreat()
            if threat then
                -- 有威胁 → 进入或更新战斗模式
                if mode == "guard" or currentEnemy ~= threat then
                    enterCombatMode(threat)
                end
                -- 更新敌方char引用（对方可能刚重生）
                currentEnemyChar = threat.Character
            else
                -- 无威胁 → 守卫模式
                if mode == "combat" then enterGuardMode() end
            end
        end

        -- ════════════════════════════
        --  守卫模式：跟随主人，不攻击
        -- ════════════════════════════
        if mode == "guard" then
            local masterRoot = masterChar and masterChar:FindFirstChild("HumanoidRootPart")
            if not masterRoot then return end

            local toMaster = masterRoot.Position - mRoot.Position
            local dist = toMaster.Magnitude
            local dir = dist > 0.05 and toMaster.Unit or -masterRoot.CFrame.LookVector
            local separation = getWhitelistSeparation(mRoot)

            camera.CFrame = CFrame.lookAt(camera.CFrame.Position, masterRoot.Position)

            if dist > CFG.masterFollowMaxDist then
                pcall(function()
                    mRoot.CFrame = CFrame.lookAt(mRoot.Position,
                        Vector3.new(masterRoot.Position.X, mRoot.Position.Y, masterRoot.Position.Z))
                end)
                local moveDirection = dir
                if separation.Magnitude > 0.05 then
                    moveDirection = moveDirection + separation.Unit
                end
                humanoid:Move(moveDirection.Magnitude > 0.05 and moveDirection.Unit or Vector3.zero, false)
                if dist > CFG.masterDashRange
                    and now - lastDashTime >= CFG.dashDebounce then
                    performDash()
                    lastDashTime = now
                end
            elseif dist < CFG.masterFollowMinDist then
                local moveDirection = -dir
                if separation.Magnitude > 0.05 then
                    moveDirection = moveDirection + separation.Unit
                end
                humanoid:Move(moveDirection.Magnitude > 0.05 and moveDirection.Unit or Vector3.zero, false)
                pcall(function()
                    mRoot.CFrame = CFrame.lookAt(mRoot.Position,
                        Vector3.new(masterRoot.Position.X, mRoot.Position.Y, masterRoot.Position.Z))
                end)
            elseif separation.Magnitude > 0.05 then
                humanoid:Move(separation.Unit, false)
            else
                humanoid:Move(Vector3.zero, false)
            end

            setStatus(string.format(
                "状态: 🛡 守卫模式\n主人: %s | %.1fm | 跟随%d-%dm",
                CFG.masterName, dist, CFG.masterFollowMinDist, CFG.masterFollowMaxDist),
                Color3.fromRGB(100, 200, 255))

        -- ════════════════════════════
        --  战斗模式：追击锁定的敌方
        -- ════════════════════════════
        elseif mode == "combat" then
            if not currentEnemyChar then return end
            local eRoot = currentEnemyChar:FindFirstChild("HumanoidRootPart")
            if not eRoot then return end

            local eDist = (eRoot.Position - mRoot.Position).Magnitude
            local eDir = eDist > 0.05
                and (eRoot.Position - mRoot.Position).Unit
                or mRoot.CFrame.LookVector
            local separation = getWhitelistSeparation(mRoot)
            local assignment, protectedMember = getArenaAssignment(currentEnemy, eRoot)
            arenaRole = assignment

            camera.CFrame = CFrame.lookAt(camera.CFrame.Position, eRoot.Position)

            -- 冷却在所有模式持续计时；达到60秒且有敌人时尝试G，成功后守卫半径变为280。
            tryActivateUltimate(now)
            updateUltimateState(now)

            if ultimateActive and not ultimateSkillsReady then
                stopAttacking()
                humanoid:Move(Vector3.zero, false)
                setStatus(string.format(
                    "状态: ⚡ 大招开启动画中\n敌方: %s | 守卫%.0fm",
                    currentEnemy.Name, getActiveGuardRadius()), Color3.fromRGB(255, 210, 80))
                return
            end

            if arenaEnabled and not trainingEnabled and assignment == "retreat" then
                local away = mRoot.Position - eRoot.Position
                if away.Magnitude < 0.05 then away = -eRoot.CFrame.LookVector end

                -- 残血位被贴近时边后撤边普攻；离开攻击范围后恢复全速撤离。
                if eDist <= CFG.attackRange and not enemyIsRagdoll then
                    if isCounterDefending() then
                        stopAttacking()
                        humanoid:Move(Vector3.zero, false)
                    else
                        local moveDirection = away.Unit
                        if separation.Magnitude > 0.05 then
                            moveDirection = moveDirection + separation.Unit
                        end
                        humanoid:Move(moveDirection.Magnitude > 0.05
                            and moveDirection.Unit or Vector3.zero, false)
                        pcall(function()
                            mRoot.CFrame = CFrame.lookAt(mRoot.Position,
                                Vector3.new(eRoot.Position.X, mRoot.Position.Y, eRoot.Position.Z))
                        end)
                        attackLoop()
                    end
                    setStatus(string.format(
                        "状态: 🩸 残血近身反击\n敌方: %s | %.1fm",
                        currentEnemy.Name, eDist), Color3.fromRGB(255, 130, 100))
                    return
                end

                stopAttacking()
                local retreatPoint = eRoot.Position + away.Unit * CFG.arenaRetreatRange
                moveTowardPoint(mRoot, retreatPoint, separation)
                if eDist < CFG.arenaRetreatRange
                    and now - lastDashTime >= CFG.dashDebounce then
                    performDash(); lastDashTime = now
                end
                setStatus(string.format(
                    "状态: 🩸 擂台撤离\n敌方: %s | %.1fm",
                    currentEnemy.Name, eDist), Color3.fromRGB(255, 110, 110))
                return
            end

            -- 敌方布娃娃：撤退
            if enemyIsRagdoll and not ultimateActive then
                stopAttacking()
                -- 50米内仅在目标布娃娃时起跳，并在离地后释放 Blitz Shot。
                local fleeDir = -eDir
                if isCounterDefending() then
                    humanoid:Move(Vector3.zero, false)
                else
                    tryRagdollAirBlitz(eDist)
                    pcall(function()
                        mRoot.CFrame = CFrame.lookAt(mRoot.Position, mRoot.Position + fleeDir)
                    end)
                    humanoid:Move(eDist < CFG.fleeDist and fleeDir or Vector3.zero, false)
                end
                setStatus(string.format(
                    "状态: ⚔ 战斗中\n敌方: %s | %.1fm | 布娃娃↩",
                    currentEnemy.Name, eDist), Color3.fromRGB(255, 180, 50))
                return
            end

            -- 敌人近距离使用指定动画：正面阶段保持实际12米并绕后，非正面即可前压攻击。
            local closeBackstepActive = isEnemyBackstepCloseActive(eRoot)
            if closeBackstepActive then
                local defending = isCounterDefending()
                local actionLocked = tick() < enemyBackstepActionUntil
                    or isSideDashActionLocked()

                if not actionLocked then
                    enemyBackstepAdvanceAttack = isOutsideEnemyFront(
                        mRoot, eRoot, enemyBackstepAdvanceAttack
                    )
                end

                if not actionLocked then
                    if enemyBackstepAdvanceAttack then
                        local attackPoint = getRadialPoint(eRoot, mRoot,
                            CFG.enemyBackstepAttackDistance)
                        local attackDelta = Vector3.new(
                            attackPoint and attackPoint.X - mRoot.Position.X or 0,
                            0,
                            attackPoint and attackPoint.Z - mRoot.Position.Z or 0
                        )
                        if attackPoint and attackDelta.Magnitude > CFG.enemyBackstepTolerance then
                            moveTowardPoint(mRoot, attackPoint, separation)
                        else
                            humanoid:Move(Vector3.zero, false)
                        end
                        if defending then
                            stopAttacking()
                        elseif eDist <= CFG.attackRange then
                            attackLoop()
                        else
                            stopAttacking()
                        end
                    else
                        stopAttacking()
                        moveAroundEnemyAtDistance(mRoot, eRoot,
                            CFG.enemyBackstepDistance, separation)
                    end

                    pcall(function()
                        mRoot.CFrame = CFrame.lookAt(mRoot.Position,
                            Vector3.new(eRoot.Position.X, mRoot.Position.Y, eRoot.Position.Z))
                        camera.CFrame = CFrame.lookAt(camera.CFrame.Position, eRoot.Position)
                    end)
                end

                if defending then
                    stopAttacking()
                elseif not actionLocked then
                    useCombatSkill(eDist)

                    if not enemyBackstepAdvanceAttack then
                        if now - lastDashTime >= CFG.dashDebounce then
                            performDash(eRoot, true)
                            lastDashTime = now
                        else
                            performSideDash(eRoot, true, true)
                        end
                    else
                        performSideDash(eRoot, true, true)
                    end
                end
                setStatus(string.format(
                    "状态: %s\n敌方: %s | %.1fm | 技能/冲刺/侧闪攻击",
                    defending and (enemyBackstepAdvanceAttack and "防御中→背侧靠近" or "防御中→正面绕后")
                        or (enemyBackstepAdvanceAttack and "背侧→靠近攻击" or "正面保持12米绕后"),
                    currentEnemy.Name, eDist), Color3.fromRGB(255, 170, 80))
                return
            end

            -- 擂台辅助位保持中远程，保护位进入敌方与残血队友之间。
            if isSideDashActionLocked() then
                -- Side-dash coroutine owns movement until its attachment step completes.
            elseif isCounterDefending() then
                stopAttacking()
                moveWhileDefending(mRoot, eRoot, separation)
            elseif arenaEnabled and not trainingEnabled and assignment == "support" then
                local away = mRoot.Position - eRoot.Position
                if away.Magnitude < 0.05 then away = eRoot.CFrame.RightVector end
                moveTowardPoint(mRoot,
                    eRoot.Position + away.Unit * CFG.arenaSupportRange, separation)
            elseif arenaEnabled and not trainingEnabled
                and assignment == "protector" and protectedMember then
                local towardProtected = protectedMember.root.Position - eRoot.Position
                if towardProtected.Magnitude < 0.05 then
                    towardProtected = -eRoot.CFrame.LookVector
                end
                moveTowardPoint(mRoot,
                    eRoot.Position + towardProtected.Unit * CFG.arenaInterceptRange, separation)
            elseif eDist > CFG.combatFollowDist then
                local pursuitPoint = eRoot.Position
                local backstepActive = isEnemyBackstepActive(eRoot)
                if backstepActive then
                    pursuitPoint = getEnemyBackstepPoint(eRoot) or pursuitPoint
                end
                pcall(function()
                    mRoot.CFrame = CFrame.lookAt(mRoot.Position,
                        Vector3.new(pursuitPoint.X, mRoot.Position.Y, pursuitPoint.Z))
                end)
                local pursuitDelta = Vector3.new(
                    pursuitPoint.X - mRoot.Position.X,
                    0,
                    pursuitPoint.Z - mRoot.Position.Z
                )
                local moveDirection = pursuitDelta.Magnitude > 0.05
                    and pursuitDelta.Unit or Vector3.zero
                if separation.Magnitude > 0.05 then
                    moveDirection = moveDirection + separation.Unit
                end
                humanoid:Move(moveDirection.Magnitude > 0.05 and moveDirection.Unit or Vector3.zero, false)
            elseif separation.Magnitude > 0.05 then
                humanoid:Move(separation.Unit, false)
            else
                humanoid:Move(Vector3.zero, false)
            end

            -- Blitz Shot 动画期间，根据目标水平移动方向保持12米提前量。
            -- 目标静止时 aimPoint 自动回到目标当前位置。
            updateBlitzPreAim(mRoot, eRoot)

            -- 冲刺
            if not isCounterDefending()
                and not isSideDashActionLocked()
                and (not arenaEnabled or trainingEnabled or assignment == "fighter")
                and eDist > CFG.combatDashMinRange
                and eDist <= CFG.combatDashRange
                and now - lastDashTime >= CFG.dashDebounce then
                performDash(eRoot, isEnemyBackstepActive(eRoot)); lastDashTime = now
            end

            -- 技能按距离决策；Blitz Shot 在其他技能不可用时全距离兜底。
            if not isCounterDefending() and not isSideDashActionLocked()
                and assignment ~= "retreat" then
                useCombatSkill(eDist)
            end

            -- 攻击 + 侧闪
            if isCounterDefending() or isSideDashActionLocked() then
                stopAttacking()
            elseif (not arenaEnabled or trainingEnabled
                or assignment == "fighter" or assignment == "protector")
                and eDist <= CFG.attackRange then
                attackLoop()
                performSideDash(eRoot, isEnemyBackstepActive(eRoot), true)
            else
                stopAttacking()
            end

            local ultTag = ultimateActive
                and (ultimateSkillsReady and " | ⚡大招" or " | ⚡启动中")
                or (ultimateTrying and " | ⚡尝试中" or "")
            local combatModeText = focusEnabled and "🎯 集中攻击" or ("⚔ 擂台" .. assignment)
            setStatus(string.format(
                "状态: %s%s\n敌方: %s | %.1fm | 守卫%.0fm",
                combatModeText, ultTag, currentEnemy.Name, eDist, getActiveGuardRadius()), Color3.fromRGB(255, 120, 80))
        end
    end)
end

-- ══════════════════════════════════════
--  本地死亡/重生
-- ══════════════════════════════════════
local function onLocalDeath()
    if not runtime.alive then return end
    disconnectAll(); stopAttacking()
    comboGeneration = comboGeneration + 1
    comboExecuting = false
    comboWindowUntil = 0
    comboMasterAnimDetected = false
    comboEnemyAnimDetected = false
    comboTarget = nil
    comboObservedEnemy = nil
    clearEnemyBackstep()
    isStandby = false
    blitzJumpPending = false
    cancelCounterAttack()
    deactivateUltimate()
    standbyAnchor = nil
    standbyAutoRotate = nil
    standbySavedParts = {}
    standbyDisplayType = nil
    setStatus("状态: 💫 重生中...", Color3.fromRGB(160, 160, 160))
    -- 不用 :Wait()，改用事件监听，避免角色已加载时永远挂起
    if localDiedConn then localDiedConn:Disconnect(); localDiedConn = nil end
    localDiedConn = LP.CharacterAdded:Connect(function(newChar)
        if not runtime.alive then return end
        if localDiedConn then localDiedConn:Disconnect(); localDiedConn = nil end
        character = newChar
        humanoid  = newChar:WaitForChild("Humanoid")
        if not runtime.alive then return end
        -- 绑定下一次死亡
        localDiedConn = humanoid.Died:Connect(onLocalDeath)
        task.wait(1)  -- 等角色物理完全加载
        if not runtime.alive then return end
        guardAndCombat()
    end)
end

-- ══════════════════════════════════════
--  热重载清理 / 启动
-- ══════════════════════════════════════
local function cleanup()
    if runtime.cleaning or runtime.cleaned then return end
    runtime.cleaning = true
    runtime.alive = false
    trackEnabled = false
    trainingEnabled = false
    comboEnabled = false
    comboExecuting = false
    comboGeneration = comboGeneration + 1
    clearEnemyBackstep()

    disconnectAll()
    if localDiedConn then
        pcall(function() localDiedConn:Disconnect() end)
        localDiedConn = nil
    end
    if guiHeartbeatConn then
        pcall(function() guiHeartbeatConn:Disconnect() end)
        guiHeartbeatConn = nil
    end

    stopAttacking()
    cancelCounterAttack()
    pcall(function()
        RunService:UnbindFromRenderStep(COMBO_CAMERA_BIND_KEY)
    end)
    sendKey(Enum.KeyCode.F, false)
    sendKey(Enum.KeyCode.W, false)
    sendKey(Enum.KeyCode.Q, false)
    sendKey(Enum.KeyCode.D, false)
    local savedAutoRotate = standbyAutoRotate
    pcall(function() exitStandby() end)
    pcall(function()
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if root then
            root.Anchored = false
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end
        if humanoid then
            humanoid:Move(Vector3.zero, false)
            humanoid.PlatformStand = false
            humanoid.AutoRotate = savedAutoRotate ~= nil and savedAutoRotate or true
        end
    end)

    pcall(function()
        local gui = LP.PlayerGui:FindFirstChild("AutoBattleGUI")
        if gui then gui:Destroy() end
    end)

    if GLOBAL_ENV[INSTANCE_KEY] == runtime then
        GLOBAL_ENV[INSTANCE_KEY] = nil
    end
    runtime.cleaning = false
    runtime.cleaned = true
end

runtime.cleanup = cleanup
